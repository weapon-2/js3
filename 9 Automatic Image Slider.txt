        <img id="img" src="https://unblast.com/wp-content/uploads/2021/01/Space-Background-Image-2.jpg" alt="">
        <button class="pre" onclick="prev()">Previous</button>
        <button class="nxt" onclick="next()">Next</button>
        <script>
            let images = ["https://unblast.com/wp-content/uploads/2021/01/Space-Background-Image-2.jpg","https://thumbs.dreamstime.com/b/incredibly-beautiful-sunset-sun-lake-sunrise-landscape-panorama-nature-sky-amazing-colorful-clouds-fantasy-design-115177001.jpg","https://png.pngtree.com/thumb_back/fh260/background/20240919/pngtree-a-background-of-orange-blue-and-yellow-gradients-with-gritty-appearance-image_16233934.jpg"]
            let ind = 0;
            function next(){
                ind++;
                if(ind==images.length){
                    ind = 0;
                }
                document.getElementById("img").src = images[ind];
            }
            function prev(){
                ind--;
                if(ind==-1){
                    ind = images.length-1
                }
                document.getElementById("img").src = images[ind];
            }
        </script>
    </div>
