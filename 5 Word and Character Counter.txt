<!DOCTYPE html>
<html>

<head>
    <title>Word and Character Counter</title>
</head>

<body>

    <h2>Live Word and Character Counter</h2>

    <textarea id="mainText" rows="6" cols="50" placeholder="Start typing here..."></textarea>

    <p>Characters: <span id="charCount">0</span></p>
    <p>Words: <span id="wordCount">0</span></p>

    <script>
        document.getElementById("mainText").addEventListener("input", function () {

            let text = this.value;

            document.getElementById("charCount").innerText = text.length;

            let trimmed = text.trim();
            let words = trimmed === "" ? 0 : trimmed.split(/\s+/).length;

            document.getElementById("wordCount").innerText = words;
        });
    </script>

</body>

</html>
